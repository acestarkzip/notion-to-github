import ctypes
import os

lib_path = os.path.abspath("./example.dll")  # 리눅스/맥은 example.so

# C++ 함수의 인자 타입(argtypes)과 반환 타입(restype) 지정
cpp_lib.add.argtypes = [ctypes.c_int, ctypes.c_int]
cpp_lib.add.restype = ctypes.c_int

# 함수 실행
result = cpp_lib.add(10, 20)
